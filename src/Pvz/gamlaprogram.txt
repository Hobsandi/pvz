        public abstract class MovingBehavior (int ticksPerStep)
        {
            private int ticksSinceStep = 0;

            public void Update(Zombie mover)
            {
                ticksSinceStep++;
                if (ticksSinceStep < ticksPerStep) return;

                ticksSinceStep = 0;
                Move(mover);
            }

            protected abstract void Move(Zombie mover);
        }



        public class WalkingBehavior(int ticksPerStep) : MovingBehavior(ticksPerStep)
        {
            public override void Move(Zombie mover)
            {
                mover.X -= PIXEL_SIZE;
            }
        }

        public class Delayer(IUpdater updater, int ticksPerUpdate) : IUpdater
        {
            int ticksSinceUpdate = 0;
            public int TicksPerUpdate { get; set; } = ticksPerUpdate;

            public void Update()
            {
                ticksSinceUpdate++;
                if (ticksSinceUpdate < TicksPerUpdate) return;

                ticksSinceUpdate = 0;
                updater.Update();
            }
